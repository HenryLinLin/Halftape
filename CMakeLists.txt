cmake_minimum_required(VERSION 3.24)
project(Halftape VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
add_library(wscore INTERFACE)
target_include_directories(wscore INTERFACE wscore/include)
add_executable(wsbench wsbench/main.cpp)
target_link_libraries(wsbench PRIVATE wscore)
enable_testing()
add_test(NAME smoke COMMAND wsbench --version)
add_executable(core_tests tests/core.cpp)
target_link_libraries(core_tests PRIVATE wscore)
add_test(NAME core COMMAND core_tests)

cmake_minimum_required(VERSION 3.24)
set(CMAKE_OSX_ARCHITECTURES arm64 CACHE STRING "Halftape supports Apple Silicon" FORCE)
set(CMAKE_OSX_DEPLOYMENT_TARGET 14.4 CACHE STRING "Minimum macOS" FORCE)
project(Halftape VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
if(NOT EXISTS "${CMAKE_SOURCE_DIR}/.deps/stretch/signalsmith-stretch.h")
  message(FATAL_ERROR "Run scripts/bootstrap.sh first")
endif()
file(GLOB DEMUCS_SOURCES CONFIGURE_DEPENDS .deps/demucs/src/*.cpp)
list(FILTER DEMUCS_SOURCES EXCLUDE REGEX "model_load.cpp$")
file(READ .deps/demucs/src/model_load.cpp MODEL_LOADER)
string(REPLACE "fread(&n_dims, sizeof(int32_t), 1, f);" "if (fread(&n_dims, sizeof(int32_t), 1, f) != 1) break;" MODEL_LOADER "${MODEL_LOADER}")
file(CONFIGURE OUTPUT "${CMAKE_BINARY_DIR}/model_load.cpp" CONTENT "${MODEL_LOADER}" @ONLY)
add_library(demucs STATIC ${DEMUCS_SOURCES} "${CMAKE_BINARY_DIR}/model_load.cpp")
target_include_directories(demucs SYSTEM PUBLIC .deps/demucs/src .deps/demucs/vendor/eigen)
target_compile_options(demucs PRIVATE -O3 -w)
if(APPLE)
  target_compile_definitions(demucs PRIVATE EIGEN_USE_BLAS)
  target_link_libraries(demucs PUBLIC "-framework Accelerate")
endif()
add_library(wscore STATIC wscore/src/separation.cpp wscore/src/analysis.cpp wscore/src/sing.cpp wscore/src/streaming.cpp)
target_include_directories(wscore PUBLIC wscore/include .deps/stretch .deps/linear/include .deps/r8brain)
target_link_libraries(wscore PUBLIC demucs)
add_executable(wsbench wsbench/main.cpp)
target_link_libraries(wsbench PRIVATE wscore)
enable_testing()
add_test(NAME smoke COMMAND wsbench --version)
add_executable(core_tests tests/core.cpp)
target_link_libraries(core_tests PRIVATE wscore)
add_test(NAME core COMMAND core_tests)
add_executable(transport_tests tests/transport.cpp)
target_link_libraries(transport_tests PRIVATE wscore)
add_test(NAME transport COMMAND transport_tests)
if(APPLE)
  enable_language(OBJCXX)
  set(CMAKE_OSX_DEPLOYMENT_TARGET 14.4)
  add_library(mactap mactap/src/platform.mm mactap/src/source_control.mm)
  target_include_directories(mactap PUBLIC mactap/include)
  target_compile_options(mactap PRIVATE -fobjc-arc)
  target_link_libraries(mactap PUBLIC wscore "-framework AppKit" "-framework CoreAudio" "-framework AudioUnit" "-framework ApplicationServices" "-framework IOKit")
  add_executable(live_device_tests tests/live_device.cpp)
  target_link_libraries(live_device_tests PRIVATE mactap)
endif()

if(APPLE)
  add_library(wsbridge STATIC wsbridge/src/session.cpp wsbridge/src/bridge.cpp)
  target_include_directories(wsbridge PUBLIC wsbridge/include)
  target_link_libraries(wsbridge PUBLIC wscore mactap)
endif()
option(WS_RTSAN "Enable Clang RealtimeSanitizer" OFF)
if(WS_RTSAN)
  include(CheckCXXCompilerFlag)
  set(CMAKE_REQUIRED_LINK_OPTIONS -fsanitize=realtime)
  check_cxx_compiler_flag("-fsanitize=realtime" HAS_RTSAN)
  unset(CMAKE_REQUIRED_LINK_OPTIONS)
  if(NOT HAS_RTSAN)
    message(FATAL_ERROR "RealtimeSanitizer requires LLVM Clang 20 or newer")
  endif()
  target_compile_options(wscore PUBLIC -fsanitize=realtime -g)
  target_link_options(wscore PUBLIC -fsanitize=realtime)
endif()
add_executable(rt_tests tests/realtime.cpp)
target_link_libraries(rt_tests PRIVATE wscore)
add_test(NAME rt_tests COMMAND rt_tests)
add_executable(analysis_tests tests/analysis.cpp)
target_link_libraries(analysis_tests PRIVATE wscore)
add_test(NAME analysis COMMAND analysis_tests)
option(WS_ONNX "Build the ONNX Runtime backend" ON)
if(WS_ONNX)
  set(ORT_ROOT "${CMAKE_SOURCE_DIR}/.deps/onnxruntime-osx-arm64-1.30.0")
  if(NOT EXISTS "${ORT_ROOT}/include/onnxruntime_cxx_api.h")
    message(FATAL_ERROR "Run scripts/bootstrap.sh for ONNX Runtime, or configure with -DWS_ONNX=OFF")
  endif()
  add_library(onnxruntime SHARED IMPORTED)
  set_target_properties(onnxruntime PROPERTIES IMPORTED_LOCATION "${ORT_ROOT}/lib/libonnxruntime.1.30.0.dylib" INTERFACE_INCLUDE_DIRECTORIES "${ORT_ROOT}/include")
  target_sources(wscore PRIVATE wscore/src/onnx.cpp)
  target_compile_definitions(wscore PRIVATE WS_ONNX)
  target_link_libraries(wscore PUBLIC onnxruntime)
endif()
add_executable(session_tests tests/session.cpp)
target_link_libraries(session_tests PRIVATE wsbridge)
add_test(NAME session COMMAND session_tests)
add_executable(capture_tests tests/capture.cpp tests/capture_platform.cpp wsbridge/src/session.cpp wsbridge/src/bridge.cpp)
target_include_directories(capture_tests PRIVATE wsbridge/include mactap/include)
target_link_libraries(capture_tests PRIVATE wscore)
set(SING_TEST_MODEL "")
if(WS_ONNX)
  set(SING_TEST_MODEL "${CMAKE_SOURCE_DIR}/.cache/models/stemgenrt-hop128.onnx")
endif()
add_test(NAME capture COMMAND capture_tests "${SING_TEST_MODEL}")
add_executable(sing_tests tests/sing.cpp)
target_link_libraries(sing_tests PRIVATE wscore)
add_test(NAME sing COMMAND sing_tests)
add_executable(streaming_tests tests/streaming.cpp)
target_link_libraries(streaming_tests PRIVATE wscore)
add_test(NAME streaming COMMAND streaming_tests "${SING_TEST_MODEL}" "${CMAKE_SOURCE_DIR}/tests/fixtures/streaming.bin")
set_tests_properties(streaming PROPERTIES SKIP_RETURN_CODE 77)

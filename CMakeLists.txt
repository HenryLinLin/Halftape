cmake_minimum_required(VERSION 3.24)
project(Halftape VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
if(NOT EXISTS "${CMAKE_SOURCE_DIR}/.deps/stretch/signalsmith-stretch.h")
  message(FATAL_ERROR "Run scripts/bootstrap.sh first")
endif()
file(GLOB DEMUCS_SOURCES CONFIGURE_DEPENDS .deps/demucs/src/*.cpp)
list(FILTER DEMUCS_SOURCES EXCLUDE REGEX "model_load.cpp$")
file(READ .deps/demucs/src/model_load.cpp MODEL_LOADER)
string(REPLACE "fread(&n_dims, sizeof(int32_t), 1, f);" "if (fread(&n_dims, sizeof(int32_t), 1, f) != 1) break;" MODEL_LOADER "${MODEL_LOADER}")
file(WRITE "${CMAKE_BINARY_DIR}/model_load.cpp" "${MODEL_LOADER}")
add_library(demucs STATIC ${DEMUCS_SOURCES} "${CMAKE_BINARY_DIR}/model_load.cpp")
target_include_directories(demucs SYSTEM PUBLIC .deps/demucs/src .deps/demucs/vendor/eigen)
target_compile_options(demucs PRIVATE -O3 -w)
if(APPLE)
  target_compile_definitions(demucs PRIVATE EIGEN_USE_BLAS)
  target_link_libraries(demucs PUBLIC "-framework Accelerate")
endif()
add_library(wscore STATIC wscore/src/separation.cpp wscore/src/analysis.cpp)
target_include_directories(wscore PUBLIC wscore/include .deps/stretch .deps/linear/include .deps/r8brain)
target_link_libraries(wscore PUBLIC demucs)
add_executable(wsbench wsbench/main.cpp)
target_link_libraries(wsbench PRIVATE wscore)
enable_testing()
add_test(NAME smoke COMMAND wsbench --version)
add_executable(core_tests tests/core.cpp)
target_link_libraries(core_tests PRIVATE wscore)
add_test(NAME core COMMAND core_tests)
if(APPLE)
  enable_language(OBJCXX)
  set(CMAKE_OSX_DEPLOYMENT_TARGET 14.4)
  add_library(mactap mactap/src/platform.mm)
  target_include_directories(mactap PUBLIC mactap/include)
  target_compile_options(mactap PRIVATE -fobjc-arc)
  target_link_libraries(mactap PUBLIC wscore "-framework AppKit" "-framework CoreAudio" "-framework AudioUnit")
endif()
